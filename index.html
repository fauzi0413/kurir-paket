1. Surface Speed Multiplier & Anti-Cheat (Go)

package game

type SurfaceType string

const (
    SurfaceAsphalt  SurfaceType = "asphalt"  // 100% Speed (0.56)
    SurfaceConcrete SurfaceType = "concrete" // 80% Speed  (0.45)
    SurfaceDirt     SurfaceType = "dirt"     // 60% Speed  (0.34)
    SurfaceGrass    SurfaceType = "grass"    // 35% Speed  (0.20)
)

func ValidatePlayerStep(p *Courier, nextX, nextZ, dt float64, surf SurfaceType) bool {
    dist := math.Hypot(nextX-p.X, nextZ-p.Z)
    maxAllowed := GetSurfaceMaxSpeed(surf) * (dt / 16.67) * 1.15
    return dist <= maxAllowed
}

2. Mutex-Safe Random Order Dispatcher (Tanpa Duplikasi Beruntun)

func (d *Dispatcher) NextTargetHouse(houses []int) int {
    d.mu.Lock()
    defer d.mu.Unlock()

    candidates := make([]int, 0, len(houses))
    for _, id := range houses {
        if id != d.lastDeliveredHouseID {
            candidates = append(candidates, id)
        }
    }
    picked := candidates[rand.Intn(len(candidates))]
    d.lastDeliveredHouseID = picked
    return picked
}
